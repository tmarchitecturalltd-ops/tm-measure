// ── Conversion tracking helper ──────────────────────────────────────────────
function trackConversion(eventName) {
  // Google Analytics 4 event
  if (typeof gtag !== 'undefined') {
    gtag('event', eventName, { event_category: 'conversion', event_label: eventName });
    // Also fire GA4 lead event for key CTAs
    if (['form_submit','hero_primary_cta','final_cta','nav_cta'].includes(eventName)) {
      gtag('event', 'generate_lead', { currency: 'GBP', event_label: eventName });
    }
  }
  // Google Tag Manager dataLayer push
  if (typeof dataLayer !== 'undefined') {
    dataLayer.push({ event: eventName, eventCategory: 'conversion' });
  }
}

function goTo(idx){
  var track = document.getElementById('reviews-track');
  var dotsEl = document.getElementById('carousel-dots');
  if (!track) return;
  var cards = track.querySelectorAll('.review-card');
  var vc = getVisibleCount();
  var max = cards.length - vc;
  carouselPos = Math.max(0, Math.min(idx, max));
  var cardW = cards[0].offsetWidth + 24;
  track.style.transform = 'translateX(-'+(carouselPos*cardW)+'px)';
  if (dotsEl){ dotsEl.querySelectorAll('button').forEach(function(d,i){ d.className = 'w-2 h-2 rounded-full transition-colors '+(i===carouselPos?'bg-primary':'bg-outline-variant'); }); }
}
function moveCarousel(dir){ goTo(carouselPos+dir); }

// ── Form submit via Formsubmit ───────────────────────────────────────────────
function sendEnquiryEmail(e){
  e.preventDefault();
  var name = document.getElementById('f-name').value.trim();
  var email = document.getElementById('f-email').value.trim();
  var location = document.getElementById('f-location').value.trim();
  var service = document.getElementById('service-select');
  var projType = document.getElementById('project-type-select');
  var message = document.getElementById('f-message').value.trim();
  var serviceText = service.options[service.selectedIndex]?.text || '—';
  var projTypeText = projType.options[projType.selectedIndex]?.text || '—';
  var btn = document.getElementById('send-btn');
  btn.textContent = 'SENDING…'; btn.disabled = true;
  trackConversion('form_submit');
  fetch('https://formsubmit.co/ajax/tmarchitecturalltd@gmail.com', {
    method:'POST',
    headers:{'Content-Type':'application/json','Accept':'application/json'},
    body: JSON.stringify({
      _subject:'New Project Enquiry – TM Designs Ltd',
      Name:name, Email:email, Location:location||'—',
      Service:serviceText, 'Project Type':projTypeText, Message:message||'—',
      _template:'table', _captcha:'false'
    })
  })
  .then(function(res){ return res.json(); })
  .then(function(){ 
    btn.classList.add('hidden'); 
    document.getElementById('form-success').classList.remove('hidden');
    document.getElementById('project-contact-form').reset(); 
    updateProjectTypes();
  })
  .catch(function(){
    var subject = encodeURIComponent('New Project Enquiry – TM Designs Ltd');
    var body = encodeURIComponent('Name: '+(name||'—')+'\nEmail: '+(email||'—')+'\nLocation: '+(location||'—')+'\nService: '+serviceText+'\nProject Type: '+projTypeText+'\n\nMessage:\n'+(message||'—'));
    window.location.href = 'mailto:tmarchitecturalltd@gmail.com?subject='+subject+'&body='+body;
    btn.textContent = 'SEND MY PROJECT DETAILS →'; btn.disabled = false;
  });
}


// ── Drag-to-scroll work gallery ────────────────────────────────────────────
(function() {
    var el = document.getElementById('workScroll');
    if (!el) return;
    var isDown = false, startX, scrollLeft;
    el.addEventListener('mousedown', function(e) {
        isDown = true; el.style.cursor = 'grabbing';
        startX = e.pageX - el.offsetLeft; scrollLeft = el.scrollLeft;
    });
    el.addEventListener('mouseleave', function() { isDown = false; el.style.cursor = 'grab'; });
    el.addEventListener('mouseup', function() { isDown = false; el.style.cursor = 'grab'; });
    el.addEventListener('mousemove', function(e) {
        if (!isDown) return;
        e.preventDefault();
        var x = e.pageX - el.offsetLeft;
        el.scrollLeft = scrollLeft - (x - startX) * 1.5;
    });
})();

    var max = cards.length - getVisibleCount();
    goTo(carouselPos >= max ? 0 : carouselPos+1);
  }, 4000);
});
</script>
